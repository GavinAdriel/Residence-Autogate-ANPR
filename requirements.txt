# ANPR Autogate System - pinned runtime + test dependencies.
# Python >=3.10 required (see pyproject.toml).

# --- Runtime ---
ultralytics==8.3.0
# Ultralytics object tracking (model.track) needs the `lap` linear-assignment
# module, provided by the `lapx` wheel. Not pulled in by the ultralytics wheel,
# so it is pinned explicitly (otherwise ultralytics auto-installs it at runtime).
lapx==0.9.4
paddleocr==2.9.1
# PaddleOCR requires the PaddlePaddle engine at runtime; it is not pulled in
# automatically by the paddleocr wheel, so it must be pinned explicitly.
# Must stay on the 2.6.x line: PaddlePaddle 3.x breaks paddleocr 2.9.1 inference
# with a OneDNN "fused_conv2d" error, so OCR silently fails at runtime.
paddlepaddle==2.6.2
opencv-python==4.10.0.84
PyQt5==5.15.11
PyYAML==6.0.2
mysql-connector-python
# Pure-Python MySQL client driver for the Docker-hosted MySQL 8.0 backend
# (replaces the previous SQLite persistence layer). Pinned exactly.
# Used by the anpr persistence layer.
PyMySQL==1.1.1
# Required by PyMySQL to complete MySQL 8.0's default caching_sha2_password
# handshake. Without it, connecting raises "'cryptography' package is required
# for sha256_password or caching_sha2_password auth methods" whenever the
# server's in-memory auth cache is cold (e.g. after a MySQL restart), so it is
# a hard runtime dependency, not an optional extra.
cryptography==43.0.3
